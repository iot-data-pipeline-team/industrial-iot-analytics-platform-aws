# Git Workflow

# Check the current status
git status

# See the current branch
git branch

# Pull the latest changes
git pull

# Stage all changes
git add .

# Stage a specific file
git add <file>

# Commit your changes
git commit -m "Describe your changes"

# Edit the previous commit message (before pushing)
git commit --amend -m "New commit message"

# Push to your branch
git push origin aws-cloud-migration-malek
